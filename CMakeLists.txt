cmake_minimum_required(VERSION 3.20)
project(trace_oar_gui LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(imgui STATIC third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp)
target_include_directories(imgui PUBLIC third_party/imgui)
add_library(trace_menu STATIC src/menu.cpp)
target_include_directories(trace_menu PUBLIC src)
target_link_libraries(trace_menu PUBLIC imgui)
target_compile_definitions(trace_menu PRIVATE TRACE_FONT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts")
option(TRACE_BUILD_PREVIEW "Build standalone SDL preview" ON)
if(TRACE_BUILD_PREVIEW)
 find_package(SDL2 REQUIRED)
 add_executable(trace_preview src/preview.cpp third_party/imgui/backends/imgui_impl_sdl2.cpp third_party/imgui/backends/imgui_impl_sdlrenderer2.cpp)
 target_include_directories(trace_preview PRIVATE third_party/imgui/backends)
 target_link_libraries(trace_preview PRIVATE trace_menu SDL2::SDL2)
 enable_testing()
 add_executable(trace_policy_test src/runtime/policy_test.cpp)
 add_test(NAME runtime_policy_test COMMAND trace_policy_test)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 find_program(TRACE_LUA_EXECUTABLE lua)
 if(Python3_FOUND AND TRACE_LUA_EXECUTABLE)
  add_test(NAME bootstrap_cache_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/run_bootstrap_test.py)
 endif()
 add_test(NAME menu_self_test COMMAND trace_preview --self-test)
 set_tests_properties(menu_self_test PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
endif()

option(TRACE_BUILD_RUNTIME "Build Win64 game DLL and injector" OFF)
if(TRACE_BUILD_RUNTIME)
 if(NOT WIN32)
  message(FATAL_ERROR "TRACE_BUILD_RUNTIME requires a Windows toolchain")
 endif()
 add_subdirectory(third_party/minhook)
 add_library(trace_game_runtime STATIC src/runtime/runtime.cpp src/runtime/ue_api.cpp)
 target_include_directories(trace_game_runtime PUBLIC src/runtime)
 target_compile_definitions(trace_game_runtime PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 target_link_libraries(trace_game_runtime PUBLIC bcrypt)
 add_library(TraceRuntime SHARED src/runtime/overlay.cpp third_party/imgui/backends/imgui_impl_win32.cpp third_party/imgui/backends/imgui_impl_dx11.cpp)
 target_include_directories(TraceRuntime PRIVATE src/runtime third_party/imgui/backends)
 target_link_libraries(TraceRuntime PRIVATE trace_menu trace_game_runtime minhook d3d11 dxgi d3dcompiler bcrypt dwmapi)
 target_compile_definitions(TraceRuntime PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 set_target_properties(TraceRuntime PROPERTIES PREFIX "")
 add_executable(TraceInject src/runtime/injector.cpp)
 target_compile_definitions(TraceInject PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 if(MINGW)
  target_link_options(TraceRuntime PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(TraceInject PRIVATE -municode -static -static-libgcc -static-libstdc++)
 endif()
endif()

if(UNIX AND NOT APPLE)
 configure_file(packaging/TraceOAR/Scripts/main.lua "${CMAKE_BINARY_DIR}/bootstrap/main.lua" COPYONLY)
 add_library(trace_loader SHARED src/loader/loader.cpp)
 set_target_properties(trace_loader PROPERTIES OUTPUT_NAME TraceLoader)
 add_executable(TraceLoader src/loader/main.cpp)
 target_link_libraries(TraceLoader PRIVATE trace_loader)
 set_target_properties(TraceLoader PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
 if(TRACE_BUILD_PREVIEW AND Python3_FOUND)
  add_test(NAME loader_integration_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/test_loader.py $<TARGET_FILE:TraceLoader>)
 endif()
endif()
